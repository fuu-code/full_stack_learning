---------------- GIT TERMINAL CODE

git init            # initiate a git control
git config --global user.email "bookeryarn@gmail.com"
git config --global user.name "Fuutarou"
git config --global init.defaultBranch main

git status          # status of files if is git

git add {file name} or git add .                # add files to git staging area 
git rm --cached -r .                            # removed all file in staging area      
git restore --staged filename.py                # remove specific file in the staging area
git commit -m "{message}"                       # commit changes
git log                                         # check the committed logs
git checkout                                    # backtrack from the previous version
git diff {file name}                            # to see the difference from old version to new version
git remote add origin {https://github.com/fuu-code/Story.git}       #link to the repository
git branch -M main                              # rename the current branch to "main"
git push -u origin main                         # upload files to github to the link


------------------- BRANCHES -------------------

git branch {name of other branch}               # create a branch name
git branch                                      # to show the branches
git branch {new-branch} {source-branch}         # create branch and move to that branch
git checkout {branch name}                      # to move to the branch
        # before merging go to {main} branch
git merge {branch name}    and type   :q!    to save and quit
git push --set-upstream origin {branch-name}    # to push branch to github
git push -u origin {branch-name}                # to push branch to github
